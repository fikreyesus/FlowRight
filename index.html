function jsonp(url, params) {
return new Promise((resolve, reject) => {
const cbName = 'jsonp_cb_' + Date.now();
const timeout = setTimeout(() => {
delete window[cbName];
document.head.removeChild(script);
reject(new Error('JSONP timeout'));
}, 10000);
window[cbName] = function(data) {
clearTimeout(timeout);
delete window[cbName];
document.head.removeChild(script);
resolve(data);
};
const qs = Object.entries({ ...params, callback: cbName })
.map(([k,v]) => k + '=' + encodeURIComponent(v)).join('&');
const script = document.createElement('script');
script.src = url + '?' + qs;
script.onerror = () => { clearTimeout(timeout); reject(new Error('JSONP error')); };
document.head.appendChild(script);
});
}

function saveDevice() {
const custId = viewingCustId;
if (!custId) return;

const data = {
name: document.getElementById('dv-name').value.trim(),
serial: document.getElementById('dv-serial').value.trim(),
type: document.getElementById('dv-type').value,
mfr: document.getElementById('dv-mfr').value.trim(),
model: document.getElementById('dv-model').value.trim(),
size: document.getElementById('dv-size').value.trim(),
loc: document.getElementById('dv-loc').value.trim(),
year: parseInt(document.getElementById('dv-year').value) || null,
tested: document.getElementById('dv-tested').value,
result: document.getElementById('dv-result').value,
notes: document.getElementById('dv-notes').value.trim(),
};
if (!data.name && !data.type) { alert('Please enter a device name or select a type.'); return; }
if (editDevId) {
const idx = deviceRecords[custId].findIndex(d => d.id === editDevId);
deviceRecords[custId][idx] = { ...data, id: editDevId };
} else {
data.id = 'dv' + Date.now();
deviceRecords[custId].push(data);
// sync count back to customer
const c = customers.find(x => x.id === custId);
if (c) c.devices = deviceRecords[custId].length;
}
closeModal('device-modal');
renderDetailBody(custId);
renderDevices();
updateHdr();
}
function delDevice(custId, devId) {

const c = customers.find(x => x.id === custId);
if (c) c.devices = deviceRecords[custId].length;
renderDetailBody(custId);
renderDevices();
updateHdr();
}
